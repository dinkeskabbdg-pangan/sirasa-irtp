function navigateTo(page) {

  document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));

  const target = document.getElementById('page-' + page);

  if (target) { target.classList.add('active'); target.classList.remove('fade-in'); void target.offsetWidth; target.classList.add('fade-in'); }

  document.getElementById('sub-nav').classList.toggle('hidden', page === 'dashboard');

  currentPage = page;

  window.scrollTo(0, 0);

  lucide.createIcons();

}

function adjustBrightness(hex, pct) {

  let r = parseInt(hex.slice(1,3),16), g = parseInt(hex.slice(3,5),16), b = parseInt(hex.slice(5,7),16);

  r = Math.min(255, r + Math.round(r * pct / 100));

  g = Math.min(255, g + Math.round(g * pct / 100));

  b = Math.min(255, b + Math.round(b * pct / 100));

  return `#${r.toString(16).padStart(2,'0')}${g.toString(16).padStart(2,'0')}${b.toString(16).padStart(2,'0')}`;

}
